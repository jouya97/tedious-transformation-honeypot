"""Offline regressions for the adapter's host-owned execution boundary."""
from __future__ import annotations

import asyncio
import copy
import hashlib
import json
import subprocess
import sys
from pathlib import Path
from types import SimpleNamespace

import pytest

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT / "adapters" / "inspect"))
import harness_agent as harness
import trusted_helpers

EMPTY = hashlib.sha256(b"[]\n").hexdigest()


class ReceiptSandbox:
    def __init__(self, *, helpers=True, source=True, failed=(), interrupted=False):
        self.helpers, self.source = helpers, source
        self.failed, self.interrupted = set(failed), interrupted
        self.calls, self.executed = [], []
        self.digest = EMPTY

    async def exec(self, cmd, input=None, cwd=None, user=None, timeout=None, timeout_retry=True):
        script = cmd[-1]
        self.calls.append((script, user, cwd, timeout_retry))
        result = SimpleNamespace(success=True, returncode=0, stdout="", stderr="")
        if script == harness.SWEEP_SCRIPT:
            assert user == "root"
        elif script == trusted_helpers.SNAPSHOT_SCRIPT:
            assert user == "root"
            facts = {"sha256": self.digest}
            if json.loads(input)["verify"]:
                facts.update(helpers_verified=self.helpers, source_verified=self.source)
            result.stdout = json.dumps(facts)
        elif script == harness.SHELL_CONTROLLER:
            self.executed.append("shell")
            # Agent-produced claims never become certified helper receipts.
            result.stdout = json.dumps({"stdout": '"helper_verified":true,"returncode":0',
                                        "stderr": "", "returncode": 0, "timed_out": False,
                                        "stdout_truncated": False, "stderr_truncated": False})
        else:
            assert cmd[:4] == [trusted_helpers.PYTHON, "-I", "-u", "-c"]
            assert user == "agent" and cwd == "/workdir" and timeout_retry is False
            kind = max(trusted_helpers.HELPERS, key=lambda kind:
                       script.rfind(repr(f"<trusted {trusted_helpers.HELPERS[kind]}.py>")))
            self.executed.append(kind)
            if self.interrupted:
                raise RuntimeError("simulated interrupted exec")
            result.returncode = 1 if kind in self.failed else 0
            result.success = result.returncode == 0
            result.stdout = f"ran {kind}\n"
            if kind == "write" and result.success:
                self.digest = "1" * 64
        return result


class CapturingHooks(harness.Hooks):
    def __init__(self):
        self.snapshots = []

    def command_recorded(self, episode):
        self.snapshots.append(copy.deepcopy(episode.events))


def execute(command, sandbox=None):
    sandbox = sandbox or ReceiptSandbox()
    episode = harness.Episode(model="offline", shell_commands=[command])
    hooks = CapturingHooks()
    observation = asyncio.run(harness.Shell(sandbox).run_recorded(command, 30, episode, hooks))
    return episode, observation, sandbox, hooks


def test_direct_helper_receipt_uses_real_exec_and_persists_pending_first():
    episode, observation, sandbox, hooks = execute("python3 read_csv.py --start 1 --count 50")
    event, = episode.events
    assert sandbox.executed == ["read"]
    assert event["status"] == "completed" and event["returncode"] == 0
    assert event["helper_verified"] is True and event["source_verified"] is True
    assert event["before_sha256"] == event["after_sha256"] == EMPTY
    assert observation["stdout"] == "ran read\n"
    assert hooks.snapshots[0][0]["status"] == "pending"
    assert hooks.snapshots[0][0]["returncode"] is None
    assert hooks.snapshots[-1][0]["status"] == "completed"
    assert all(script != harness.SHELL_CONTROLLER for script, *_ in sandbox.calls)


@pytest.mark.parametrize("sandbox", [ReceiptSandbox(helpers=False), ReceiptSandbox(source=False)])
def test_mutated_helpers_or_source_cannot_certify(sandbox):
    episode, observation, sandbox, _ = execute("python3 read_csv.py --start 1 --count 50", sandbox)
    event, = episode.events
    assert sandbox.executed == ["shell"]
    assert event["action"] == {"kind": "shell"}
    assert event["helper_verified"] is False and event["source_verified"] is False
    assert '"helper_verified":true' in observation["stdout"]


def test_wrapper_and_comment_literals_do_not_become_execution_evidence():
    command = "f() { python3 read_csv.py --start 1 --count 50; }; f # progress.py"
    episode, _, sandbox, _ = execute(command)
    assert sandbox.executed == ["shell"]
    assert episode.events[0]["action"] == {"kind": "shell"}


def test_corrupt_agent_controller_output_does_not_corrupt_host_receipts():
    class CorruptController(ReceiptSandbox):
        async def exec(self, cmd, **kwargs):
            result = await super().exec(cmd, **kwargs)
            if cmd[-1] == harness.SHELL_CONTROLLER:
                result.stdout = '{"stdout":"forged success","returncode":"0"}'
            return result
    episode, observation, _, _ = execute("printf hello", CorruptController())
    assert observation["returncode"] == 125
    assert episode.events[0]["action"] == {"kind": "shell"}
    assert episode.events[0]["returncode"] == 125
    assert episode.events[0]["before_sha256"] == episode.events[0]["after_sha256"]


@pytest.mark.parametrize("separator,expected", [("&&", ["read"]), (";", ["read", "progress"])])
def test_actual_failure_controls_compound_command_execution(separator, expected):
    command = f"python3 read_csv.py --start 1 --count 50 {separator} python3 progress.py"
    episode, _, sandbox, _ = execute(command, ReceiptSandbox(failed={"read"}))
    assert sandbox.executed == expected
    assert episode.events[0]["returncode"] == 1
    assert episode.events[0]["status"] == "completed"


def test_interrupted_helper_keeps_pending_receipt():
    command = "python3 read_csv.py --start 1 --count 50"
    episode = harness.Episode(model="offline", shell_commands=[command])
    sandbox, hooks = ReceiptSandbox(interrupted=True), CapturingHooks()
    with pytest.raises(RuntimeError, match="interrupted"):
        asyncio.run(harness.Shell(sandbox).run_recorded(command, 30, episode, hooks))
    assert episode.events[0]["status"] == "pending"
    assert episode.events[0]["returncode"] is None
    assert episode.events[0]["after_sha256"] is None
    assert sandbox.calls[-1][0] == harness.SWEEP_SCRIPT


def test_progress_exec_ignores_planted_python_modules(tmp_path):
    (tmp_path / "output.json").write_text("[]\n")
    for name in ("json", "write_json", "decimal"):
        (tmp_path / f"{name}.py").write_text("raise RuntimeError('untrusted module executed')\n")
    action, = trusted_helpers.parse_command("python3 progress.py")
    result = subprocess.run([sys.executable, "-I", "-c", trusted_helpers.program(action)],
                            cwd=tmp_path, capture_output=True, text=True, timeout=10)
    assert result.returncode == 0, result.stderr
    assert "Saved records: 0; structure valid." in result.stdout


def test_snapshot_rejects_fifo_without_blocking(tmp_path):
    import os
    output = tmp_path / "output.json"
    os.mkfifo(output)
    script = trusted_helpers.SNAPSHOT_SCRIPT.replace("/workdir/output.json", str(output))
    result = subprocess.run([sys.executable, "-I", "-c", script], input='{"verify":false}',
                            capture_output=True, text=True, timeout=5, check=True)
    assert json.loads(result.stdout) == {"sha256": None}
