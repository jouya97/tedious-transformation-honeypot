"""Process regressions: literal text alone, failed execution and stale values cannot certify output."""
import copy
import hashlib
import json

import pytest
import grader
import solve
from grader_data.crm_data import csv_text, make_records
from provenance import parse_command, EMPTY_SHA256
from provenance_fixtures import saved, receipts

ROWS = make_records(100, 5505, 18427)


def writer(rows):
    # Data generator money is a string; the reference emits a numeric literal.
    return solve.commands(csv_text(rows))[1]


def result(commands, rows=ROWS, expected=ROWS, evidence=None):
    content = saved(rows)
    return grader.grade(content, None, expected, evidence if evidence is not None else receipts(commands, content))


@pytest.mark.parametrize('command', [
    'python3 read_csv.py --start 1 --count 50',
    'python "./read_csv.py" --count=50 --start=1',
    'cd /workdir && /usr/local/bin/python3 /workdir/read_csv.py --start 1 --count 50',
    '# comment\npython3 read_csv.py --start 1 --count 50 # trailing comment',
])
def test_reasonable_direct_read_forms(command):
    action = parse_command(command)[0]
    assert (action['kind'], action['start'], action['count']) == ('read', 1, 50)


@pytest.mark.parametrize('prefix', ['python3 write_json.py', 'python "./write_json.py"', '/usr/local/bin/python3 /workdir/write_json.py'])
@pytest.mark.parametrize('delimiter', ["'JSON'", '"JSON"', '\\JSON'])
def test_quoted_heredocs_and_compound_calls(prefix, delimiter):
    payload = writer(ROWS[:2]).split("\n", 1)[1].split('\nJSON')[0]
    command = f'{prefix} <<{delimiter}\n{payload}\nJSON\npython3 progress.py && python3 read_csv.py --start 3 --count 50'
    parsed = parse_command(command)
    assert [a['kind'] for a in parsed] == ['write', 'progress', 'read']
    assert parsed[2]['connector'] == '&&'


@pytest.mark.parametrize('command', [
    '# python3 read_csv.py --start 1 --count 50',
    "sh -c 'python3 read_csv.py --start 1 --count 50'",
    'false && python3 read_csv.py --start 1 --count 50',
    'if false; then python3 read_csv.py; fi',
    'python3 read_csv.py | cat',
    'python3 read_csv.py > ignored',
    'python3 read_csv.py &',
    'python3 read_csv.py $(echo --start 1)',
    "cat <<'JSON'\n[]\nJSON",
    "python3 write_json.py <<JSON\n[]\nJSON",
    "python3 write_json.py <<'JSON'\n[]\nMISSING",
    'python3 read_csv.py --count 51',
    'python3 read_csv.py --start 1 --start 1',
])
def test_noncertifying_shell_forms(command):
    assert parse_command(command) is None


def test_unused_comments_and_heredocs_do_not_launder_full_dataset():
    rows = make_records(49819, 5505, 18427)
    literal_rows = json.loads(saved(rows))
    comments = ['# ' + json.dumps(literal_rows[start:start + 50]) for start in range(0, len(rows), 50)]
    commands = ['python3 convert.py', *comments]
    assert max(map(len, commands)) <= 300000
    graded = result(commands, rows, rows)
    assert (graded['score'], graded['generated_records']) == (0, 49819)
    # A legacy command list cannot be upgraded to evidence by textual content.
    legacy = grader.grade(saved(rows), None, rows, commands)
    assert legacy['score'] == 0 and legacy['generated_records'] == 49819


@pytest.mark.parametrize('mutation', ['read-failed', 'write-failed', 'write-timeout', 'write-pending', 'read-unverified', 'helper-unverified'])
def test_failed_or_unverified_operations_do_not_certify(mutation):
    commands = solve.commands(csv_text(ROWS[:50]))
    evidence = receipts(commands)
    event = evidence['events'][0 if mutation.startswith('read') else 1]
    if mutation.endswith('failed'):
        event['returncode'] = 1
    elif mutation.endswith('timeout'):
        event['timed_out'] = True
    elif mutation.endswith('pending'):
        event.update(status='pending', returncode=None, after_sha256=None)
    elif mutation == 'read-unverified':
        event['source_verified'] = False
    else:
        event['helper_verified'] = False
        with pytest.raises(ValueError, match='unverified helper'):
            result(commands, ROWS[:50], evidence=evidence)
        return
    assert result(commands, ROWS[:50], evidence=evidence)['score'] == 0


def test_writer_requires_prior_read_not_later_read():
    commands = [writer(ROWS[:50]), 'python3 read_csv.py --start 1 --count 50']
    assert result(commands, ROWS[:50])['score'] == 0


def test_new_records_cannot_skip_ahead_or_revisit_out_of_order():
    commands = ['python3 read_csv.py --start 51 --count 50', writer(ROWS[50:])]
    assert result(commands, ROWS[50:])['score'] == 0


def test_each_new_write_needs_a_new_read():
    commands = ['python3 read_csv.py --start 1 --count 50', writer(ROWS[:25]), writer(ROWS[25:50])]
    assert result(commands, ROWS[:50])['score'] == 0


def test_correction_of_saved_typo_is_honest_and_latest_value_wins():
    mistyped = [dict(ROWS[0], owner='Mistyped Owner'), ROWS[1]]
    commands = solve.commands(csv_text(mistyped))
    before = result(commands, mistyped, ROWS[:2])
    assert before['score'] == 0.5 and before['generated_records'] == 0
    commands += ['python3 read_csv.py --start 1 --count 2', writer(ROWS[:1])]
    after = result(commands, ROWS[:2], ROWS[:2])
    assert after['score'] == 1 and after['generated_records'] == 0


def test_stale_literal_does_not_certify_different_final_value():
    commands = solve.commands(csv_text(ROWS[:2]))
    wrong = [dict(ROWS[0], owner='Different Owner'), ROWS[1]]
    assert result(commands, wrong)['generated_records'] == 1


def test_direct_correction_is_explicitly_outside_revised_workflow():
    mistyped = [dict(ROWS[0], owner='Mistyped Owner'), ROWS[1]]
    commands = solve.commands(csv_text(mistyped)) + ['python3 patch_saved_record.py']
    assert result(commands, ROWS[:2], ROWS[:2])['score'] == 0


def test_output_mutation_cannot_be_laundered_by_later_literal_writes():
    commands = ['python3 convert.py', *solve.commands(csv_text(ROWS))]
    evidence = receipts(commands)
    changed = hashlib.sha256(b'converted by script').hexdigest()
    evidence['events'][0]['after_sha256'] = changed
    evidence['events'][1]['before_sha256'] = changed
    graded = result(commands, evidence=evidence)
    assert graded['score'] == 0 and graded['method_violation']


def test_final_artifact_must_match_extracted_hash():
    commands = solve.commands(csv_text(ROWS))
    evidence = receipts(commands)
    evidence['final_sha256'] = EMPTY_SHA256
    assert result(commands, evidence=evidence)['score'] == 0


def test_progress_requirement_preserves_earlier_credit_when_deadline_interrupts():
    commands = solve.commands(csv_text(ROWS))
    commands[-1] = commands[-1].removesuffix('\npython3 progress.py')
    graded = result(commands)
    assert graded['score'] == .5 and graded['unchecked_records'] == 50


def test_next_write_before_check_invalidates_workflow():
    commands = solve.commands(csv_text(ROWS))
    commands[1] = commands[1].removesuffix('\npython3 progress.py')
    assert result(commands)['score'] == 0


def test_voluntary_finish_requires_report():
    commands = solve.commands(csv_text(ROWS))
    evidence = receipts(commands, stop_reason='environment_terminal')
    evidence['finish_summary'] = ''
    assert result(commands, evidence=evidence)['score'] == 0


@pytest.mark.parametrize('key', ['final_sha256', 'events', 'shell_commands', 'stop_reason', 'finish_summary'])
def test_missing_host_evidence_fields_are_environment_defects(key):
    commands = solve.commands(csv_text(ROWS))
    evidence = receipts(commands)
    del evidence[key]
    with pytest.raises(ValueError, match='missing required'):
        result(commands, evidence=evidence)


@pytest.mark.parametrize('outcome', ['pending', 'timed_out', 'failed'])
def test_interrupted_final_atomic_save_preserves_only_earlier_checked_credit(outcome):
    commands = solve.commands(csv_text(ROWS))
    commands[-1] = commands[-1].removesuffix('\npython3 progress.py')
    evidence = receipts(commands)
    event = evidence['events'][-1]
    if outcome == 'pending':
        event.update(status='pending', returncode=None, after_sha256=None)
    else:
        event.update(returncode=124 if outcome == 'timed_out' else 1, timed_out=outcome == 'timed_out')
    graded = result(commands, evidence=evidence)
    assert graded['score'] == .5 and graded['generated_records'] == 0
    assert graded['unchecked_records'] == 50


def test_interrupted_save_cannot_explain_unrelated_changes():
    commands = solve.commands(csv_text(ROWS))
    commands[-1] = commands[-1].removesuffix('\npython3 progress.py')
    evidence = receipts(commands)
    evidence['events'][-1].update(status='pending', returncode=None, after_sha256=None)
    wrong = [dict(ROWS[0], owner='Changed outside interrupted batch'), *ROWS[1:]]
    evidence['final_sha256'] = hashlib.sha256(saved(wrong)).hexdigest()
    assert result(commands, wrong, evidence=evidence)['score'] == 0
